2 Log-Odds Ratio (Uninformative Dirichlet Prior)

Fightin’ Words (Monroe, Colaresi & Quinn, 2008): z = log((y_f + a)/(n_f - y_f + a)) - log((y_o + a)/(n_o - y_o + a)) with the symmetric prior a on every count. Positive z = word is more likely to appear in a failed trace; negative z = more likely in a correct trace. Words shared across both classes (boilerplate, prompt-leaked labels) sit near z = 0 and drop out of the extremes.

<script>
(function () {
  var statusEl = document.getElementById('phrase-net-status');
  if (typeof vis === 'undefined' || !vis.Network) {
    if (statusEl) statusEl.textContent = 'Network library failed to load — see static SVG below.';
    return;
  }
  var base = (window.location.pathname.match(/.*\//) || ['/'])[0];
  fetch(base + '../charts/phrase_net_differential.json')
    .then(function (r) { if (!r.ok) throw new Error('HTTP ' + r.status); return r.json(); })
    .then(function (data) {
      var nodeMap = {};
      data.nodes.forEach(function (n) { nodeMap[n.id] = n; });
      var nodes = data.nodes.map(function (n) {
        var size = 14 + 6 * Math.log1p(n.fail);
        var share = Math.max(0, Math.min(1, n.share));
        var lo = [0.12, 0.62, 0.35], hi = [0.84, 0.27, 0.27];
        var c = [0, 1, 2].map(function (i) { return lo[i] * (1 - share) + hi[i] * share; });
        var hex = '#' + c.map(function (v) { return Math.round(v * 255).toString(16).padStart(2, '0'); }).join('');
        return {
          id: n.id, label: n.id, size: size, color: { background: hex, border: '#333333' },
          shape: n.leak ? 'diamond' : 'circle',
          title: n.id + '<br>failed traces: <b>' + n.fail + '</b> · correct: <b>' + n.ok + '</b>' +
                 '<br>failure share: <b>' + (100 * share).toFixed(1) + '%</b>' +
                 (n.leak ? '<br><i>prompt-leaked vocabulary</i>' : ''),
          font: { color: '#1b2a4a', size: 13, face: 'DejaVu Sans' },
          borderWidth: 1.2, borderWidthSelected: 2.5, shadow: { enabled: true, size: 12 }
        };
      });
      var zMax = Math.max.apply(null, data.edges.map(function (e) { return e.z; }));
      var edges = data.edges.map(function (e) {
        return {
          from: e.from, to: e.to,
          width: 0.6 + 4.4 * e.z / zMax,
          color: e.in_cycle ? { color: '#d64545', highlight: '#b03333', opacity: 1.0 } : { color: '#a0a8b8', opacity: 0.55 },
          arrows: { to: { enabled: true, scaleFactor: 0.75 } },
          title: e.from + ' → ' + e.to + '<br>z = <b>' + e.z.toFixed(2) + '</b>' +
                 '<br>failed: <b>' + e.fail + '</b> · correct: <b>' + e.ok + '</b>' +
                 (e.in_cycle ? '<br><i>structural loop</i>' : ''),
          smooth: { type: 'continuous', roundness: 0.25 }
        };
      });
      var container = document.getElementById('phrase-net-canvas');
      var opts = {
        physics: { enabled: true, stabilization: { iterations: 300 }, barnesHut: { gravitationalConstant: -32000, springLength: 130, springConstant: 0.035, damping: 0.6 } },
        interaction: { hover: true, tooltipDelay: 120, dragNodes: true, zoomView: true, hoverConnectedEdges: true, hideEdgesOnDrag: true },
        nodes: { scaling: { min: 8, max: 42 } },
        edges: { selectionWidth: 2, hoverWidth: 2 }
      };
      var network = new vis.Network(container, { nodes: nodes, edges: edges }, opts);
      network.once('stabilizationIterationsDone', function () {
        network.setOptions({ physics: false });
        if (statusEl) statusEl.textContent = data.nodes.length + ' words · ' + data.edges.length + ' edges · drag, zoom, hover';
      });
      network.on('stabilizationProgress', function (p) {
        if (statusEl) statusEl.textContent = 'Laying out graph… ' + Math.round(100 * p.iterations / p.total) + '%';
      });
    })
    .catch(function (err) {
      if (statusEl) statusEl.textContent = 'Could not load graph data (' + err.message + ') — see static SVG below.';
    });
})();
</script>
